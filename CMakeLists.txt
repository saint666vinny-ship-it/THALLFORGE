cmake_minimum_required(VERSION 3.22)
project(THALLFORGE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
FetchContent_Declare(SoundTouch GIT_REPOSITORY https://github.com/RTE-Dev/soundtouch.git GIT_TAG main GIT_SHALLOW TRUE)
set(SOUNDSTRETCH OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(SoundTouch)
juce_add_gui_app(THALLFORGE PRODUCT_NAME "THALLFORGE")
target_sources(THALLFORGE PRIVATE Source/Main.cpp)
target_compile_definitions(THALLFORGE PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="THALLFORGE")
target_link_libraries(THALLFORGE PRIVATE juce::juce_audio_utils juce::juce_dsp SoundTouch)
